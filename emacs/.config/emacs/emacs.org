* Overview & Architecture

This literate configuration is the single source of truth for Emacs 31+ across
Linux and macOS, supporting both graphical frames (=emacsclient -c=) and
terminal frames inside Tmux (=emacsclient -t= under Kitty, WezTerm, or Alacritty).

Saving this file automatically tangles and byte-compiles its outputs:
- =early-init.el= — GC tuning, native compilation, Truecolor environment setup,
  Evil pre-init flags, =package-quickstart= cache path, and frame defaults.
- =init.el= — Package archive setup (=package.el= + built-in =use-package= with
  =:vc=), completion, polyglot IDE tooling, and coding-agent integrations.
- =user-lisp/danish-display.el= — Per-frame GUI/TUI adaptation, Truecolor face
  sync, hairline split dividers, margins, CSI-u key decoding, mouse support,
  and OSC 52 / OS clipboard parity.
- =user-lisp/danish-editing.el= — Editing and window-management utilities.
- =user-lisp/danish-modeline.el= — Hook-cached custom modeline with Nerd Icons
  and Catppuccin Macchiato styling.
- =user-lisp/danish-agents.el= — Dedicated side-window rules and context/diagnostic
  dispatchers for Claude Code, Pi, and OpenCode.

* Core Architecture & Startup

** Early Initialization (=early-init.el=)

Runs before package activation and frame creation to maximize startup speed,
guarantee 24-bit Truecolor negotiation for daemon TTY clients, prevent
white-screen flashes, and align =package-quickstart-file= with =no-littering=.

#+begin_src emacs-lisp :tangle early-init.el :comments no
;;; early-init.el --- Early initialization -*- lexical-binding: t; -*-

;;; Commentary:
;; Executed prior to `package.el' and GUI/TUI frame initialization.
;; Configures GC thresholds, native compilation, Truecolor daemon environment,
;; Evil pre-autoload flags, `package-quickstart' paths, and frame defaults.

;;; Code:

;; Defer garbage collection during startup; restore a balanced threshold once ready.
(setq gc-cons-threshold most-positive-fixnum
      gc-cons-percentage 0.6)

(add-hook 'emacs-startup-hook
          (lambda ()
            (setq gc-cons-threshold (* 32 1024 1024)
                  gc-cons-percentage 0.1)))

;; Ensure daemon TTY clients negotiate 24-bit Truecolor (16,777,216 colors)
;; before terminal initialization runs.
(unless (getenv "COLORTERM")
  (setenv "COLORTERM" "truecolor"))

;; Native compilation settings (Emacs 31+)
(when (featurep 'native-compile)
  (setq native-comp-jit-compilation t
        native-comp-async-report-warnings-errors 'silent
        native-comp-speed 2))

;; Must be set before `package-quickstart' activates Evil autoloads and hooks.
(defvar evil-mode-buffers nil)
(setq evil-want-integration t
      evil-want-keybinding nil)

;; Align quickstart cache path with `no-littering' (`var/package-quickstart.el').
(setq package-quickstart-file
      (expand-file-name "var/package-quickstart.el" user-emacs-directory)
      package-enable-at-startup t
      package-quickstart t)

;; Suppress GUI toolbars/scrollbars and set Catppuccin Macchiato defaults early.
(setq frame-inhibit-implied-resize t)
(push '(menu-bar-lines . 0) default-frame-alist)
(push '(tool-bar-lines . 0) default-frame-alist)
(push '(vertical-scroll-bars) default-frame-alist)
(push '(font . "Maple Mono NF-14") default-frame-alist)
(push '(background-color . "#24273a") default-frame-alist)
(push '(foreground-color . "#cad3f5") default-frame-alist)

(provide 'early-init)
;;; early-init.el ends here
#+end_src

** Package Archives & Built-in =use-package= (=init.el=)

Configures GNU ELPA, NonGNU ELPA, and MELPA priorities alongside Emacs 31's
built-in =use-package= (including =:vc= support). On a fresh installation where
packages are not yet present, archive metadata is fetched once automatically.

#+begin_src emacs-lisp :tangle init.el :comments no
;;; init.el --- Main Emacs configuration -*- lexical-binding: t; -*-

;;; Commentary:
;; Primary configuration entry point tangled from `emacs.org'.
;; Targets Emacs 31+ across Linux and macOS in daemon, GUI, and TUI modes.

;;; Code:

;; Increase subprocess read chunk size (1 MB) for LSP and terminal throughput.
(setq read-process-output-max (* 1024 1024))

;; Package archives and priority order
(require 'package)
(setq package-archives
      '(("gnu"    . "https://elpa.gnu.org/packages/")
        ("nongnu" . "https://elpa.nongnu.org/nongnu/")
        ("melpa"  . "https://melpa.org/packages/"))
      package-archive-priorities
      '(("gnu"    . 99)
        ("nongnu" . 80)
        ("melpa"  . 50)))

(unless package--initialized
  (package-initialize))

;; Built-in `use-package' defaults
(require 'use-package)
(setq use-package-always-ensure t
      use-package-always-defer t
      use-package-vc-prefer-newest t
      use-package-compute-statistics t)

;; Refresh archive metadata automatically only on a clean bootstrap when core
;; packages are not yet installed.
(when (and (not package-archive-contents)
           (not (package-installed-p 'no-littering)))
  (package-refresh-contents))
#+end_src

Keeps =custom.el=, backups, auto-saves, history files, and package caches

(defun danish-refresh-user-lisp ()
  "Byte-compile `user-lisp/' and regenerate `user-lisp-autoloads.el'."
  (interactive)
  (let* ((dir (locate-user-emacs-file "user-lisp"))
         (out (expand-file-name "user-lisp-autoloads.el" dir)))
    (when (file-directory-p dir)
      (add-to-list 'load-path dir)
      (byte-recompile-directory dir 0)
      (loaddefs-generate dir out)
      (load out nil 'nomessage))))

(defun danish--tangle-on-save ()
  "Automatically tangle `emacs.org' and refresh `user-lisp/' on save."
  (when (and buffer-file-name
             (string-equal (file-truename buffer-file-name)
                           (file-truename (locate-user-emacs-file "emacs.org"))))
    (let ((org-confirm-babel-evaluate nil))
      (org-babel-tangle)
      (danish-refresh-user-lisp))))

(add-hook 'after-save-hook #'danish--tangle-on-save)

;; Emacs 31 natively activates `user-lisp/' during interactive/daemon startup;
;; in `--batch' runs, load existing autoloads or generate them if missing.
(let* ((user-lisp-dir (locate-user-emacs-file "user-lisp"))
       (autoloads-file (expand-file-name "user-lisp-autoloads.el" user-lisp-dir)))
  (when (file-directory-p user-lisp-dir)
    (add-to-list 'load-path user-lisp-dir)
    (unless (fboundp 'danish-display-init)
      (if (file-exists-p autoloads-file)
          (load autoloads-file nil 'nomessage)
        (danish-refresh-user-lisp)))))
#+end_src

* Situationally Aware GUI & TUI Architecture

** Per-Frame Display, Tmux CSI-u & Clipboard Module (=user-lisp/danish-display.el=)

Adapts each frame dynamically as clients connect or switch focus—so one daemon
can simultaneously serve graphical frames (=Maple Mono NF-14=, pixel scrolling,
window fringes, native GUI clipboard) and Tmux TUI frames (24-bit Truecolor
faces, Unicode hairline split dividers =│=, 1-column fringe-equivalent margins,
xterm mouse, CSI-u decoding for =C-.= / =C-;= / =C-S-d=, margin diff indicators,
and OSC 52 + =pbcopy=/=wl-copy=/=xclip= sync).

#+begin_src emacs-lisp :tangle user-lisp/danish-display.el :comments no
;;; danish-display.el --- Situational GUI/TUI frame, input & clipboard setup -*- lexical-binding: t; -*-

;;; Commentary:
;; Dynamically adapts frame font, scrolling, Truecolor faces, split dividers,
;; margins, diff-hl indicators, CSI-u terminal key decoding, mouse support, and
;; clipboard integration per frame type.

;;; Code:

(defvar xterm-extra-capabilities)
(declare-function diff-hl-margin-mode "diff-hl" (&optional arg))
(declare-function tty-set-up-initial-frame-faces "faces" ())

(defgroup danish-display nil
  "Situational GUI and TUI display settings."
  :group 'environment)

(defcustom danish-gui-font "Maple Mono NF-14"
  "Default font specification for graphical frames."
  :type 'string
  :group 'danish-display)

(defun danish-display--setup-tui-ui (&optional frame)
  "Apply Truecolor, GUI-parity borders/margins, mouse, and CSI-u keys to TUI FRAME."
  (with-selected-frame (or frame (selected-frame))
    (unless (display-graphic-p)
      (unless (getenv "COLORTERM")
        (setenv "COLORTERM" "truecolor"))
      (when (fboundp 'tty-set-up-initial-frame-faces)
        (tty-set-up-initial-frame-faces))
      (when (featurep 'catppuccin-theme)
        (enable-theme 'catppuccin))

      ;; Replace ASCII '|' split dividers with continuous Unicode box-drawing lines
      (unless standard-display-table
        (setq standard-display-table (make-display-table)))
      (set-display-table-slot standard-display-table 'vertical-border (make-glyph-code ?│))
      (set-face-attribute 'vertical-border (selected-frame)
                          :foreground "#363a4f" :background 'unspecified)

      ;; Give TUI windows breathing room matching GUI fringes
      (set-window-margins nil 1 1)

      ;; Enable xterm mouse clicks, wheel scrolling, and drag selection
      (xterm-mouse-mode 1)
      (unless (fboundp 'mouse-wheel-mode)
        (global-set-key [mouse-4] (lambda () (interactive) (scroll-down 3)))
        (global-set-key [mouse-5] (lambda () (interactive) (scroll-up 3))))

      ;; Decode CSI-u sequences forwarded by Kitty/WezTerm/Alacritty through Tmux
      (let ((map input-decode-map))
        (define-key map "\e[46;5u" (kbd "C-."))
        (define-key map "\e[59;5u" (kbd "C-;"))
        (define-key map "\e[68;6u" (kbd "C-S-d"))
        (define-key map "\e[100;6u" (kbd "C-S-d"))
        (define-key map "\e[13;2u" (kbd "S-<return>"))
        (define-key map "\e[13;5u" (kbd "C-<return>"))))))

(defun danish-display--osc52-copy (text)
  "Copy TEXT to the outer terminal clipboard using OSC 52 (Tmux-aware)."
  (let* ((b64 (base64-encode-string (encode-coding-string text 'utf-8) t))
         (seq (if (getenv "TMUX")
                  (format "\ePtmux;\e\e]52;c;%s\a\e\\" b64)
                (format "\e]52;c;%s\a" b64))))
    (send-string-to-terminal seq)))

(defun danish-display--tui-cut-function (text)
  "Copy TEXT from TUI frames via OSC 52 and native OS clipboard utilities."
  (danish-display--osc52-copy text)
  (cond
   ((and (eq system-type 'darwin) (executable-find "pbcopy"))
    (let ((process-connection-type nil))
      (let ((proc (start-process "pbcopy" nil "pbcopy")))
        (process-send-string proc text)
        (process-send-eof proc))))
   ((and (getenv "WAYLAND_DISPLAY") (executable-find "wl-copy"))
    (let ((process-connection-type nil))
      (let ((proc (start-process "wl-copy" nil "wl-copy")))
        (process-send-string proc text)
        (process-send-eof proc))))
   ((and (getenv "DISPLAY") (executable-find "xclip"))
    (let ((process-connection-type nil))
      (let ((proc (start-process "xclip" nil "xclip" "-selection" "clipboard")))
        (process-send-string proc text)
        (process-send-eof proc))))))

(defun danish-display--sync-frame-context (&optional frame)
  "Apply GUI or TUI display, margin, and clipboard settings to FRAME."
  (let ((f (or frame (selected-frame))))
    (when (frame-live-p f)
      (with-selected-frame f
        (if (display-graphic-p f)
            (progn
              (when (find-font (font-spec :name danish-gui-font))
                (set-frame-font danish-gui-font nil t))
              (when (fboundp 'pixel-scroll-precision-mode)
                (pixel-scroll-precision-mode 1))
              (set-window-margins nil 0 0)
              (when (bound-and-true-p diff-hl-margin-mode)
                (diff-hl-margin-mode -1))
              (setq interprogram-cut-function #'gui-select-text))
          (danish-display--setup-tui-ui f)
          (when (fboundp 'diff-hl-margin-mode)
            (diff-hl-margin-mode 1))
          (setq interprogram-cut-function #'danish-display--tui-cut-function))))))

;;;###autoload
(defun danish-display-init ()
  "Register frame lifecycle hooks for situational GUI and TUI adaptation."
  (setq xterm-extra-capabilities '(modifyOtherKeys getSelection setSelection)
        save-interprogram-paste-before-kill t)
  (add-hook 'after-make-frame-functions #'danish-display--sync-frame-context)
  (add-hook 'tty-setup-hook #'danish-display--setup-tui-ui)
  (add-hook 'window-selection-change-functions
            (lambda (_win)
              (danish-display--sync-frame-context (selected-frame))))
  (unless (daemonp)
    (danish-display--sync-frame-context (selected-frame))))

(provide 'danish-display)
;;; danish-display.el ends here
#+end_src

;;; danish-editing.el --- Editing and window utilities -*- lexical-binding: t; -*-

;;; Commentary:
;; Helper commands for duplicating lines/regions, copying balanced expressions,
;; and toggling window maximization without losing split layouts.

;;;###autoload
(defun danish-duplicate-line-or-region ()
  "Duplicate the active region or the current line, preserving column."
  (interactive)
  (if (use-region-p)
      (let* ((beg (region-beginning))
             (end (region-end))
             (text (buffer-substring-no-properties beg end)))
        (goto-char end)
        (insert text))
    (let ((col (current-column))
          (line (buffer-substring-no-properties
                 (line-beginning-position) (line-end-position))))
      (end-of-line)
      (newline)
      (insert line)
      (move-to-column col))))

;;;###autoload
(defun danish-copy-sexp ()
  "Copy the following balanced expression to the kill ring without moving point."
  (interactive)
  (save-excursion
    (let ((beg (point)))
      (forward-sexp)
      (kill-ring-save beg (point)))))

;;;###autoload
(defun danish-toggle-maximize-window ()
  "Toggle maximizing the selected window while preserving the frame layout."
  (interactive)
  (if (= 1 (length (window-list)))
      (jump-to-register '_danish-max-win)
    (window-configuration-to-register '_danish-max-win)
    (delete-other-windows)))

(provide 'danish-editing)

  "Compute Flymake error/warning/note counts in a single pass and cache."

        (when (bound-and-true-p flymake-mode)
          (let ((err 0) (warn 0) (note 0)
                (sev-err (flymake--severity :error))
                (sev-warn (flymake--severity :warning)))
            (dolist (d (flymake-diagnostics))
              (let ((s (flymake--severity (flymake-diagnostic-type d))))
                (cond
                 ((>= s sev-err) (cl-incf err))
                 ((>= s sev-warn) (cl-incf warn))
                 (t (cl-incf note)))))

                  (propertize (format "%s%d " (danish-modeline--glyph "󰈸 " "E:") err)
                              'face 'error))

                              'face 'warning))
                (when (> note 0)
                  (propertize (format "%s%d " (danish-modeline--glyph " " "I:") note)
                              'face 'success)))
               'mouse-face 'mode-line-highlight
               'local-map danish-modeline--flymake-map
               'help-echo "mouse-1: buffer diagnostics | mouse-2: project diagnostics"))))))

  (defvar-local danish--oxlint-proc nil)

  (defun danish-flymake-oxlint (report-fn &rest _args)
    "Flymake backend running `oxlint' on the current buffer."
    (unless (executable-find "oxlint")
      (error "Cannot find `oxlint' executable"))
    (when (process-live-p danish--oxlint-proc)
      (kill-process danish--oxlint-proc))
    (let* ((source (current-buffer))
           (fname (or (buffer-file-name) "stdin.ts")))
      (save-restriction
        (widen)
        (setq danish--oxlint-proc
              (make-process
               :name "danish-oxlint"
               :noquery t
               :connection-type 'pipe
               :buffer (generate-new-buffer " *oxlint*")
               :command (list "oxlint" "--format" "unix" "--stdin-filepath" fname)
               :sentinel
               (lambda (proc _event)
                 (when (memq (process-status proc) '(exit signal))
                   (unwind-protect
                       (when (with-current-buffer source
                               (eq proc danish--oxlint-proc))
                         (with-current-buffer (process-buffer proc)
                           (goto-char (point-min))
                           (let (diags)
                             (while (re-search-forward
                                     "^.+?:\\([0-9]+\\):\\([0-9]+\\): \\(.+\\) \\[\\(Error\\|Warning\\)/\\(.+?\\)\\]$"
                                     nil t)
                               (let* ((line (string-to-number (match-string 1)))
                                      (col  (string-to-number (match-string 2)))
                                      (msg  (format "[%s] %s" (match-string 5) (match-string 3)))
                                      (type (if (string= (match-string 4) "Error") :error :warning))
                                      (region (flymake-diag-region source line col)))
                                 (push (flymake-make-diagnostic
                                        source (car region) (cdr region) type msg)
                                       diags)))
                             (funcall report-fn (nreverse diags)))))
                     (kill-buffer (process-buffer proc)))))))
        (process-send-region danish--oxlint-proc (point-min) (point-max))
        (process-send-eof danish--oxlint-proc))))

    "Enable `danish-flymake-oxlint' when `oxlint' is available."

(defun danish-agents--project-root ()
  "Return the current project root or `default-directory'."
  (if-let* ((proj (project-current)))
      (project-root proj)
    default-directory))

(defun danish-agents--launch-eat-cli (name cmd)
  "Launch or focus a project-scoped `eat' session named NAME running CMD."
  (require 'eat)
  (let* ((root (danish-agents--project-root))
         (proj-name (file-name-nondirectory (directory-file-name root)))
         (buf-name (format "*%s-agent:%s*" name proj-name))
         (default-directory root))
    (if-let* ((existing (get-buffer buf-name)))
        (pop-to-buffer existing)
      (let ((buf (eat cmd)))
        (with-current-buffer buf
          (rename-buffer buf-name t)
          (setq-local mode-line-format nil))
        (pop-to-buffer buf)))))

(defun danish-agents--active-agent-buffer ()
  "Return the most recently used live coding agent buffer, or nil."
  (seq-find
   (lambda (buf)
     (with-current-buffer buf
       (or (derived-mode-p 'agent-shell-mode)
           (string-match-p "\\*\\(claude\\|pi\\|opencode\\)-agent:"
                           (buffer-name buf)))))
   (buffer-list)))

(defun danish-agents--dispatch-text (payload)
  "Send PAYLOAD string to the active coding agent buffer or copy to kill-ring."
  (if-let* ((target (danish-agents--active-agent-buffer)))
      (with-current-buffer target
        (cond
         ((derived-mode-p 'agent-shell-mode)
          (goto-char (point-max))
          (insert payload)
          (pop-to-buffer target))
         ((derived-mode-p 'eat-mode)
          (when (bound-and-true-p eat-terminal)
            (eat-term-send-string eat-terminal payload))
          (pop-to-buffer target))
         ((derived-mode-p 'vterm-mode)
          (vterm-send-string payload)
          (pop-to-buffer target))))
    (kill-new payload)
    (message "No active agent buffer; copied context to clipboard/kill-ring.")))

;;;###autoload
(defun danish-agent-send-context ()
  "Send active region or current file path to the active coding agent.
When a region is active, formats it as a fenced code block with file and
line-range metadata."
  (interactive)
  (let* ((root (danish-agents--project-root))
         (rel-file (if buffer-file-name
                       (file-relative-name buffer-file-name root)
                     (buffer-name)))
         (payload
          (if (use-region-p)
              (let* ((beg (region-beginning))
                     (end (region-end))
                     (l1 (line-number-at-pos beg))
                     (l2 (line-number-at-pos end))
                     (lang (string-remove-suffix "-ts-mode"
                                                 (string-remove-suffix "-mode"
                                                                       (symbol-name major-mode))))
                     (code (buffer-substring-no-properties beg end)))
                (format "\nIn `@%s` (lines %d-%d):\n```%s\n%s\n```\n"
                        rel-file l1 l2 lang code))
            (format " `@%s` " rel-file))))
    (danish-agents--dispatch-text payload)))

;;;###autoload
(defun danish-agent-send-diagnostics ()
  "Send current buffer Flymake diagnostics to the active coding agent."
  (interactive)
  (if (not (bound-and-true-p flymake-mode))
      (user-error "Flymake is not active in this buffer")
    (let* ((root (danish-agents--project-root))
           (rel-file (if buffer-file-name
                         (file-relative-name buffer-file-name root)
                       (buffer-name)))
           (diags (flymake-diagnostics)))
      (if (null diags)
          (message "No Flymake diagnostics in %s" rel-file)
        (let ((lines
               (mapcar (lambda (d)
                         (format "- Line %d [%s]: %s"
                                 (line-number-at-pos (flymake-diagnostic-beg d))
                                 (flymake-diagnostic-type d)
                                 (flymake-diagnostic-text d)))
                       diags)))
          (danish-agents--dispatch-text
           (format "\nPlease fix these diagnostics in `@%s`:\n%s\n"
                   rel-file (string-join lines "\n"))))))))

(defun danish-agents-init ()
  "Register resilient side-window display rules for coding agent buffers."
  (add-to-list 'display-buffer-alist
               `("\\*\\(?:agent-shell\\|claude-agent\\|pi-agent\\|opencode-agent\\).*\\*"
                 (display-buffer-in-side-window)
                 (side . right)
                 (slot . 0)
                 (window-width . ,danish-agent-window-width)
                 (window-parameters . ((no-delete-other-windows . t))))))

(provide 'danish-agents)
